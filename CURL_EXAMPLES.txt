// Exemplos de Requisições com cURL
// Use esses comandos no terminal para testar a API

// ============================================
// 1. REGISTRAR NOVO USUÁRIO
// ============================================

curl -X POST http://localhost:3000/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{
    "email": "joao@exemplo.com",
    "username": "joao_silva",
    "password": "senha123",
    "passwordConfirm": "senha123"
  }'

// Resposta esperada:
// {
//   "message": "Usuário registrado com sucesso! ✅",
//   "user": {
//     "id": 1,
//     "email": "joao@exemplo.com",
//     "username": "joao_silva",
//     "is_verified": false,
//     "created_at": "2024-01-15T10:30:00.000Z"
//   },
//   "tokens": {
//     "accessToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
//     "refreshToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
//   }
// }

// ============================================
// 2. FAZER LOGIN
// ============================================

curl -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "joao@exemplo.com",
    "password": "senha123"
  }'

// Resposta esperada: (mesmo da saída acima)

// ============================================
// 3. OBTER PERFIL (PROTEGIDO)
// ============================================

curl -X GET http://localhost:3000/api/auth/profile \
  -H "Authorization: Bearer SEU_ACCESS_TOKEN_AQUI"

// Resposta esperada:
// {
//   "message": "Perfil carregado com sucesso!",
//   "user": {
//     "id": 1,
//     "email": "joao@exemplo.com",
//     "username": "joao_silva",
//     "is_verified": false,
//     "created_at": "2024-01-15T10:30:00.000Z"
//   }
// }

// ============================================
// 4. RENOVAR ACCESS TOKEN
// ============================================

curl -X POST http://localhost:3000/api/auth/refresh-token \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "SEU_REFRESH_TOKEN_AQUI"
  }'

// Resposta esperada:
// {
//   "message": "Token atualizado com sucesso!",
//   "tokens": {
//     "accessToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
//     "refreshToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
//   }
// }

// ============================================
// 5. SOLICITAR RESET DE SENHA
// ============================================

curl -X POST http://localhost:3000/api/auth/request-password-reset \
  -H "Content-Type: application/json" \
  -d '{
    "email": "joao@exemplo.com"
  }'

// Resposta esperada:
// {
//   "message": "Se o email existe, um link de reset foi enviado"
// }

// ============================================
// 6. RESETAR SENHA
// ============================================

curl -X POST http://localhost:3000/api/auth/reset-password \
  -H "Content-Type: application/json" \
  -d '{
    "token": "TOKEN_RECEBIDO_NO_EMAIL",
    "newPassword": "novaSenha123",
    "passwordConfirm": "novaSenha123"
  }'

// Resposta esperada:
// {
//   "message": "Senha resetada com sucesso! ✅",
//   "user": {
//     "id": 1,
//     "email": "joao@exemplo.com",
//     "username": "joao_silva"
//   }
// }

// ============================================
